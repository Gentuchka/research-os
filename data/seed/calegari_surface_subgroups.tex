\begin{conjecture}[D.~Calegari]
Let $G$ be a hyperbolic group. Then every rational $2$-dimensional homology class is virtually represented by a sum of surface subgroups. More precisely, for every
\[
\alpha \in H_2(G;\mathbb{Q}),
\]
there exist a finite collection of closed oriented surfaces $S_i$ and injective homomorphisms
\[
\rho_i : \pi_1(S_i) \to G
\]
such that
\[
\sum_i [S_i] = n\alpha,
\]
where $[S_i]$ denotes the image of the fundamental class of $S_i$ in $H_2(G)$.
\end{conjecture}
